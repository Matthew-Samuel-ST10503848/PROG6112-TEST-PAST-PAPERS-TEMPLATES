========================
QUESTION 2: OOP REPORT
========================

INTERFACE
public interface IExample {
    String getName();
    int getValue();
}

ABSTRACT CLASS
public abstract class Example
        implements IExample {

    private String name;
    private int value;

    public Example(String name, int value) {
        this.name = name;
        this.value = value;
    }

    @Override
    public String getName() {
        return name;
    }

    @Override
    public int getValue() {
        return value;
    }
}

SUBCLASS
public class ExampleReport extends Example {

    public ExampleReport(String name, int value) {
        super(name, value);
    }

    public void printReport() {
        System.out.println("Name: " + getName());
        System.out.println("Value: " + getValue());
    }
}

MAIN
public class RunApplication {

    public static void main(String[] args) {

        ExampleReport report =
                new ExampleReport("Example", 100);

        report.printReport();
    }
}

========================
KEY WORDS TO REMEMBER
========================

implements = class follows an interface
extends = class inherits from another class
super(...) = calls the parent constructor
this.variable = object's own variable
@Override = method is implementing/overriding a parent/interface method
new = creates an object
get(i) = gets an ArrayList value
size() = number of ArrayList elements
length = number of array elements
